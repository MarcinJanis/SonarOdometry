def frame_to_cart_tensor(frame, r_min, r_max, theta_max, device):
    """Sonar frame tensor -> Cartesian grayscale tensor (1,1,H8,W8) padded to multiples of 8."""
    frame_np = frame.detach()[0, 0, 0].cpu().numpy()         # (H, W) float [0,1]
    h_pol, w_pol = frame_np.shape
    out_shape = (h_pol, 2 * h_pol)

    frame_u8_3ch = cv2.cvtColor((frame_np * 255).astype(np.uint8), cv2.COLOR_GRAY2BGR)
    cart_bgr     = img_polar2cart(frame_u8_3ch, r_min, r_max, theta_max, out_shape=out_shape, bg=0)
    cart_gray    = cv2.cvtColor(cart_bgr, cv2.COLOR_BGR2GRAY).astype(np.float32) / 255.0

    H, W = cart_gray.shape
    H8 = ((H + 7) // 8) * 8
    W8 = ((W + 7) // 8) * 8

    padded = np.zeros((H8, W8), dtype=np.float32)
    padded[:H, :W] = cart_gray

    mask = np.zeros((H8, W8), dtype=np.float32)
    mask[:H, :W] = 1.0

    tensor = torch.from_numpy(padded).unsqueeze(0).unsqueeze(0).to(device)  # (1,1,H8,W8)
    mask_t = torch.from_numpy(mask).unsqueeze(0).to(device)                  # (1,H8,W8)
    return tensor, mask_t, cart_bgr, (H, W)

# ── Estimate R, t from LoFTR matches (2-D planar motion via RANSAC) ──────────
#
# Sonar scans a horizontal 2-D plane → rigid body motion: rotation θ + translation (tx, ty).
# Pipeline:
#   1. pixel coords  →  physical coords [m]   (using sonar geometry)
#   2. cv2.estimateAffinePartial2D (RANSAC)  →  2×3 matrix M
#   3. decompose M  →  θ [deg], tx [m], ty [m]
#   4. accumulate global poses and plot trajectory

# physical scale: 1 pixel = scale metres
_d0     = frames_data[0]
_out_h  = _d0['orig_hw'][0]
_out_w  = _d0['orig_hw'][1]
_scale  = (r_max - r_min) / _out_h   # metres per pixel


def px_to_phys(pts_px, out_h, out_w, scale, r_min):
    """Cartesian pixel (x,y) → physical (x_m, y_m) [m], matches img_polar2cart geometry."""
    x_m = (pts_px[:, 0] - out_w / 2) * scale
    y_m = (out_h - pts_px[:, 1])      * scale + r_min
    return np.stack([x_m, y_m], axis=1)

        mkpts1 = out['keypoints1'].cpu().numpy()

        good  = conf >= conf_thresh
        mkpts0, mkpts1 = mkpts0[good], mkpts1[good]

        H, W = d1['orig_hw']

        # pixel → physical [m]
        pts1_m = px_to_phys(mkpts0, H, W, _scale, r_min)
        pts2_m = px_to_phys(mkpts1, H, W, _scale, r_min)

        # RANSAC rigid 2-D transform
        M, inlier_mask = cv2.estimateAffinePartial2D(
            pts1_m, pts2_m,
            method=cv2.RANSAC,
            ransacReprojThreshold=0.15,   # [m]
            maxIters=3000,
            confidence=0.999,
        )

        if M is not None and inlier_mask is not None:
            inlier_mask = inlier_mask.ravel().astype(bool)
            theta = np.arctan2(M[1, 0], M[0, 0])
            tx, ty = M[0, 2], M[1, 2]
            n_in  = inlier_mask.sum()

            # 2-D rotation matrix  (3×3 homogeneous)

                               [ np.sin(theta),  np.cos(theta)]])

        else:

            theta, tx, ty, n_in = 0.0, 0.0, 0.0, 0
            R_mat = np.eye(2)

            frames =(start + pair_idx * stride, start + (pair_idx + 1) * stride),
            R=R_mat, t=t_vec,
            theta_deg=np.degrees(theta), tx=tx, ty=ty,
            n_matches=len(mkpts0), n_inliers=int(n_in),
        ))

        ax.scatter(mkpts1[in_mask, 0] + w1,   mkpts1[in_mask, 1],        c='lime',  s=10, zorder=4)
        ax.scatter(mkpts0[~in_mask, 0],        mkpts0[~in_mask, 1],       c='red',   s=6,  zorder=3, alpha=0.5)
        ax.scatter(mkpts1[~in_mask, 0] + w1,  mkpts1[~in_mask, 1],       c='red',   s=6,  zorder=3, alpha=0.5)

import pypose as pp

# ── TUNE: axis mapping from sonar image frame to GT robot frame ───────────────
#
# Sonar image physical coords:  x_r = right,  y_r = forward  (range direction)
# Typical robot/GT frame:       x   = forward, y   = left
#

coord_map  = np.array([[ 0, -1], [-1,  0]])   # ← change here
theta_sign = -1                                 # ← +1 or -1 (flip yaw if needed)

# ── collect GT absolute poses ─────────────────────────────────────────────────
gt_poses = []
with torch.no_grad():
    for i in range(frames_num):
        _, _, pose_gt, _ = data_generator.get_sample(start + i * stride, return_visu=True)
        gt_poses.append(pose_gt.cpu())   # SE3: [tx, ty, tz, qx, qy, qz, qw]

# ── relative GT poses: T_rel = T1^{-1} @ T2 ─────────────────────────────────
def yaw_from_se3(t):
    qx, qy, qz, qw = t[3], t[4], t[5], t[6]
    return float(torch.atan2(2*(qw*qz + qx*qy), 1 - 2*(qy**2 + qz**2)))

gt_rel = []
for i in range(len(gt_poses) - 1):
    rel = (pp.SE3(gt_poses[i]).Inv() @ pp.SE3(gt_poses[i + 1])).tensor()
    gt_rel.append(dict(tx=float(rel[0]), ty=float(rel[1]),
                       theta_deg=float(np.degrees(yaw_from_se3(rel)))))

# ── apply coord_map to estimated values ──────────────────────────────────────
def apply_map(pr, cm, ts):
    t_mapped = cm @ np.array([pr['tx'], pr['ty']])
    return dict(tx=t_mapped[0], ty=t_mapped[1],
                theta_deg=ts * pr['theta_deg'])

pair_mapped = [apply_map(pr, coord_map, theta_sign) for pr in pair_results]

# ── dead-reckoning ────────────────────────────────────────────────────────────
def dead_reckon(rel_list, key_tx='tx', key_ty='ty', key_th='theta_deg'):
    pose = np.zeros(3)
    traj = [pose.copy()]
    for r in rel_list:
        c, s = np.cos(pose[2]), np.sin(pose[2])
        pose[:2] += np.array([[c,-s],[s,c]]) @ np.array([r[key_tx], r[key_ty]])
        pose[2]  += np.radians(r[key_th])
        traj.append(pose.copy())
    return np.array(traj)
